from azure.ai.evaluation import evaluator

@evaluator
def tom_contoso(response: str) -> dict:
    """Tom de voz Contoso: formal-amigável, sem menosprezar concorrente."""
    issues, score = [], 1.0
    if any(b in response.lower() for b in ["horrível", "pior", "lixo"]):
        issues.append("Menciona concorrente de forma negativa")
        score -= 0.5
    if len(response) < 30:
        issues.append("Resposta muito curta para tom corporativo")
        score -= 0.2
    return {"score": max(0, score), "passed": score >= 0.7, "issues": issues}
